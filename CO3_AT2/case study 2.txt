# 1. POS TAGGING
# -----------------------------------------

print("1. POS TAGGING")
print("------------------------------")

sentence1 = [
    ("Book", "VB"),
    ("a", "DT"),
    ("flight", "NN"),
    ("ticket", "NN"),
    ("now", "RB")
]

sentence2 = [
    ("This", "DT"),
    ("book", "NN"),
    ("is", "VBZ"),
    ("interesting", "JJ")
]

print("Sentence 1:")
for word, tag in sentence1:
    print(word, "/", tag)

print()

print("Sentence 2:")
for word, tag in sentence2:
    print(word, "/", tag)

print()

print("Explanation:")
print("'Book' is a Verb (VB) in Sentence 1 because")
print("it represents an action.")
print()
print("'book' is a Noun (NN) in Sentence 2 because")
print("it refers to a thing/object.")

# -----------------------------------------
# 2. HMM PROBABILITY
# -----------------------------------------

print()
print("2. HMM TAGGING")
print("------------------------------")

p_book_given_vb = 0.6
p_book_given_nn = 0.4

p_start_vb = 0.5
p_start_nn = 0.5

# HMM likelihood
vb_probability = p_start_vb * p_book_given_vb
nn_probability = p_start_nn * p_book_given_nn

print("P(book | VB) =", p_book_given_vb)
print("P(book | NN) =", p_book_given_nn)

print("P(Start -> VB) =", p_start_vb)
print("P(Start -> NN) =", p_start_nn)

print()
print("VB score =", vb_probability)
print("NN score =", nn_probability)

if vb_probability > nn_probability:
    print("HMM Prediction: book = VB")
else:
    print("HMM Prediction: book = NN")

# -----------------------------------------
# 3. RULE-BASED VS HMM
# -----------------------------------------

print()
print("3. RULE-BASED VS HMM TAGGING")
print("------------------------------")

print("Rule-Based Tagging:")
print("- Uses manually written grammar rules.")
print("- Easy to understand.")
print("- Requires many rules for large datasets.")
print()

print("Stochastic / HMM Tagging:")
print("- Uses probabilities from training data.")
print("- Handles ambiguity using context.")
print("- Suitable for large-scale applications.")
print()

print("Recommendation:")
print("HMM-based tagging is more suitable for")
print("large-scale chatbot deployment because it")
print("can learn probabilities and handle ambiguity.")

# -----------------------------------------

print()
print("Examples:")

print("NN  = Noun")
print("JJ  = Adjective")
print("RB  = Adverb")
print("DT  = Determiner")
print("VBZ = Verb, 3rd person singular")

print()
print("POS tagging helps the chatbot to:")
print("1. Understand user intent.")
print("2. Identify actions and objects.")
print("3. Generate better responses.")
print("4. Resolve word ambiguity.")
