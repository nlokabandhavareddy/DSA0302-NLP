import math

print("==============================================")
print("NEWS ANALYTICS AND POS TAG CORRECTION SYSTEM")
print("==============================================")
# -----------------------------------------
# 1. INITIAL POS TAGS
# -----------------------------------------

print("1. INITIAL POS TAGS")
print("------------------------------")

sentence = [
    ("economic", "JJ"),
    ("growth", "NN"),
    ("increases", "NNS"),
    ("employment", "NN")
]

for word, tag in sentence:
    print(word, "/", tag)

# -----------------------------------------
# Apply Transformation Rule
# NNS -> VBZ if preceding word is NN
# -----------------------------------------

print()
print("Applying Transformation Rule...")
print("Rule: Change NNS to VBZ if preceding word is NN")
print()

corrected_sentence = []

for i in range(len(sentence)):

    word = sentence[i][0]
    tag = sentence[i][1]

    if i > 0:
        previous_tag = sentence[i - 1][1]

        if tag == "NNS" and previous_tag == "NN":
            tag = "VBZ"

    corrected_sentence.append((word, tag))

print("2. CORRECTED POS TAGS")
print("------------------------------")

for word, tag in corrected_sentence:
    print(word, "/", tag)

# -----------------------------------------
# 2. EXPLANATION
# -----------------------------------------

print()
print("Linguistic Explanation:")
print("'growth' is a singular noun (NN).")
print("'increases' acts as the main verb.")
print("Therefore, 'increases' should be tagged as VBZ.")
print("'employment' is a noun (NN).")

# -----------------------------------------
# 3. WORD FREQUENCY DISTRIBUTION
# -----------------------------------------

print()
print("3. WORD FREQUENCY DISTRIBUTION")
print("------------------------------")

frequencies = {
    "economic": 120,
    "growth": 450,
    "increases": 210,
    "employment": 380
}

total_frequency = sum(frequencies.values())

print("Total frequency =", total_frequency)
print()

for word, count in frequencies.items():

    probability = count / total_frequency

    print(word)
    print("Frequency =", count)
    print("Probability =", round(probability, 4))
    print()

# -----------------------------------------
# 4. ENTROPY
# -----------------------------------------

print("4. ENTROPY ANALYSIS")
print("------------------------------")

entropy = 0

for word, count in frequencies.items():

    probability = count / total_frequency

    entropy -= probability * math.log2(probability)

print("Entropy =", round(entropy, 4), "bits")

print()
print("Interpretation:")
print("Entropy measures uncertainty in the word")
print("frequency distribution.")

# -----------------------------------------
# Tagging Accuracy
# -----------------------------------------

print()
print("5. TRANSFORMATION-BASED TAGGING")
print("------------------------------")

initial_tags = ["JJ", "NN", "NNS", "NN"]
corrected_tags = ["JJ", "NN", "VBZ", "NN"]

changes = 0

for i in range(len(initial_tags)):

    if initial_tags[i] != corrected_tags[i]:
        changes += 1

print("Initial tags :", initial_tags)
print("Corrected tags:", corrected_tags)
print("Number of corrections =", changes)

print()
print("Transformation-Based Tagging improves")
print("accuracy by applying learned grammatical")
print("rules to incorrect initial tags.")
