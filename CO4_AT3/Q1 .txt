# CFG Tree vs Dependency Parsing

sentence = "The student reads a book"

print("===== CFG TREE =====")
print("Sentence")
print(" |")
print(" +-- NP")
print(" |    +-- The")
print(" |    +-- student")
print(" |")
print(" +-- VP")
print("      +-- reads")
print("      +-- NP")
print("           +-- a")
print("           +-- book")

print()
print("===== DEPENDENCY PARSING =====")
print("reads -> student  (subject)")
print("reads -> book     (object)")
print("book  -> a        (determiner)")
print("student -> The    (determiner)")

print()
print("===== COMPARISON =====")
print("CFG       : Represents hierarchical phrase structure")
print("Dependency: Represents word-to-word relationships")
print("Better for word relationships: Dependency Parsing")
