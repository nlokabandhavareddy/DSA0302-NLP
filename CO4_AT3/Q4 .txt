# Feature Structures vs Subcategorization Frames


print("===== FEATURE STRUCTURES =====")


subject = {
    "word": "She",
    "number": "singular",
    "person": "third"
}


verb = {
    "word": "runs",
    "number": "singular",
    "person": "third"
}


print("Subject:", subject)
print("Verb   :", verb)


if subject["number"] == verb["number"] and \
   subject["person"] == verb["person"]:
    print("Agreement: Correct")
else:
    print("Agreement: Incorrect")


print()
print("===== SUBCATEGORIZATION FRAME =====")


print("Verb: give")
print("Frame: Subject + Verb + Object + Recipient")
print("Example: The teacher gives a book to the student")


print()
print("===== COMPARISON =====")
print("Feature Structures      : Handle grammatical features")
print("Subcategorization Frames: Handle verb argument requirements")
print("Best for agreement      : Feature Structures")
print("Best for verb arguments : Subcategorization Frames")
Output
