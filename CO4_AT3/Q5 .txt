# Transition-Based vs Graph-Based Dependency Parsing


sentence = "The student reads a book"


print("===== TRANSITION-BASED PARSING =====")
print("Uses a sequence of parsing actions")
print("SHIFT -> READS")
print("SHIFT -> STUDENT")
print("REDUCE -> SUBJECT")
print("SHIFT -> BOOK")
print("REDUCE -> OBJECT")
print("Fast and suitable for large datasets")


print()
print("===== GRAPH-BASED PARSING =====")
print("Creates possible dependency relationships")
print("Evaluates possible dependency trees")
print("Selects the highest-scoring tree")
print("Usually more computationally expensive")


print()
print("===== COMPARISON =====")
print("Transition-Based : Fast, sequential decisions")
print("Graph-Based      : Global tree optimization")
print("Speed             : Transition-Based is faster")
print("Global accuracy   : Graph-Based can consider more relationships")
print("Best for large-scale applications: Transition-Based Parsing")
